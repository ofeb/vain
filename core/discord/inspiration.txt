def _build_embed(payload, *, page: int = 1, total: int = 1) -> discord.Embed:
    description = payload.caption or payload.title or payload.source_url
    embed = discord.Embed(description=description[:4096] if description else None)
    if payload.author_name or payload.author_username:
        name = payload.author_name or payload.author_username
        if payload.author_username and payload.author_name:
            name = f"{payload.author_name} (@{payload.author_username})"
        embed.set_author(name=name, icon_url=payload.author_avatar or None)
    footer_bits: list[str] = []
    if payload.likes:
        footer_bits.append(f"❤️ {payload.likes:,}")
    if payload.comments:
        footer_bits.append(f"💬 {payload.comments:,}")
    if payload.shares:
        footer_bits.append(f"🔗 {payload.shares:,}")
    if payload.views:
        footer_bits.append(f"👀 {payload.views:,}")
    if total > 1:
        footer_bits.append(f"Page {page}/{total}")
    if footer_bits:
        embed.set_footer(text="  ".join(footer_bits))
    return embed


async def _send_payload(
    ctx: "Context",
    payload: RepostPayload,
    *,
    allow_url_fetch: bool = False,
    url_headers: dict[str, str] | None = None,
) -> bool:
    posted = False
    media = _downloaded_media(payload)
    if not media:
        if allow_url_fetch and payload.media:
            session = aiohttp.ClientSession(timeout=aiohttp.ClientTimeout(total=30))
            try:
                item = payload.media[0]
                file = await _media_file_from_url(
                    session,
                    item,
                    headers=url_headers,
                )
                await ctx.send(embed=_build_embed(payload), file=file)
                return True
            finally:
                await session.close()

        await ctx.fail("No media could be resolved for that URL.")
        return False

    try:
        if len(media) == 1:
            file = _media_file_from_data(media[0])
            await ctx.send(embed=_build_embed(payload), file=file)
            posted = True
            return True

        pages: list[MediaPage] = []
        emojis = PaginatorEmojis(
            previous=ctx.bot.glyphs.page_previous,
            next=ctx.bot.glyphs.page_next,
            navigate=ctx.bot.glyphs.page_navigate,
            cancel=ctx.bot.glyphs.page_cancel,
        )
        for index, item in enumerate(media, start=1):
            file = _media_file_from_data(item)
            pages.append(
                MediaPage(
                    embed=_build_embed(payload, page=index, total=len(media)),
                    files=[file],
                )
            )
        paginator = MediaPaginator(
            pages,
            author_id=ctx.author.id,
            bot=ctx.bot,
            emojis=emojis,
        )
        await paginator.start(ctx.channel)
        posted = True
        return True
    except Exception:
        if posted:
            log.warning(
                "Repost for %r was posted but follow-up work failed",
                payload.source_url or media[0].filename,
                exc_info=True,
            )
            return True
        raise
