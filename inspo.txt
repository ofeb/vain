class Player(Player):
    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)

        self.queue = Queue()
        self.ctx: Context = None
        self.loop: bool = False
        self.current_track: Track = None
        self.awaiting = False

    def shuffle(self) -> None:
        return shuffle(self.queue)

    async def set_pause(self, pause: bool) -> Coroutine[Any, Any, bool]:
        if pause is True:
            self.awaiting = True
        else:
            if self.awaiting:
                self.awaiting = False

        return await super().set_pause(pause)

    async def next(self, track: Track = None) -> None:
        if not self.loop:
            if not track:
                try:
                    track: Track = self.queue.get()
                except QueueEmpty:
                    return await self.kill()

            self.current_track = track

        await self.play(self.current_track)
        await self.channel.edit(status=f"Listening to {track.title}")
        embed = Embed(
            description=f"Now playing [**{track.title}**]({track.uri}) - **{track.author}**",
            color=CONFIG.colors.base,
        )
        await self.context.send(embed=embed)

        if self.awaiting:
            self.awaiting = False

    def set_context(self, ctx: Context):
        self.context = ctx

    async def kill(self) -> Message:
        try:
            self.queue.clear()
            return await self.context.approve("Cleared the current queue.")
        except (HTTPException, KeyError) as e:
            return await self.context.approve(f"Failed to clear the queue: {str(e)}")

    async def clean_title_search(self, title: str, artist: str) -> tuple[str, str]:
        patterns = [
            r"\[.*?\]",
            r"\(from .*?\)",
            r"\(Official.*?\)",
            r"\(feat\..*?\)",
            r"\(ft\..*?\)",
            r"\(Explicit\)",
            r"\(Official Video\)",
            r"\(Audio\)",
            r"\(Lyrics\)",
            r"\(Official Visualizer\)",
            r"\(Official Music Video\)",
            r"\(Official Audio\)",
            r"\(Visualizer\)",
        ]

        clean_title = title
        for pattern in patterns:
            clean_title = sub(pattern, "", clean_title, flags=IGNORECASE)

        if " - " in clean_title:
            parts = clean_title.split(" - ", 1)
            if len(parts) == 2:
                artist, clean_title = parts

        clean_title = clean_title.strip()
        artist = artist.strip()

        logger.info(f"Cleaned search query: '{clean_title}' by '{artist}'")
        return clean_title, artist

    @command(name="stop")
    async def stop(self, ctx: Context):
        """Stops the player in the voice channel."""
        player: Player = ctx.voice_client
        await player.stop()
        player.queue.clear()
        await ctx.voice_client.disconnect()

    @command(name="play", aliases=["p"])
    async def play(self, ctx: Context, *, query: str):
        """Play a song."""
        if not ctx.author.voice or not ctx.author.voice.channel:
            return await ctx.warn(
                "You need to be in a voice channel to use this command."
            )

        player = ctx.voice_client or await ctx.author.voice.channel.connect(
            cls=Player, self_deaf=True
        )
        player.set_context(ctx)
        player.awaiting = True

        try:
            results = await player.get_tracks(query=query, ctx=ctx)
        except:
            return await ctx.warn(f"There was an issue fetching the track.")
        if not results:
            return await ctx.warn(f"No song found")

        if isinstance(results, Playlist):
            for track in results.tracks:
                player.queue.put(track)
            track = None
            await ctx.neutral(f"Added **{len(results.tracks)}** songs to the queue")

        else:
            track = results[0]
            if player.is_playing:
                player.queue.put(track)
                return await ctx.neutral(
                    f"Added [**{track.title}**]({track.uri}) to the queue"
                )

        if not player.is_playing:
            player._current = track
            await player.next(track)

    @command(name="queue", aliases=["q"])
    async def queue(self, ctx: Context):
        """
        Show the current queue.
        """
        player: Player = ctx.voice_client
        playing = f"{'Looping' if player.loop else 'Playing'} [***{player.current.title}***]({player.current.uri}) by **{player.current.author}** - **`{player.position // 60000}:{(player.position // 1000) % 60:02d}`**`/`**`{player.current.length // 60000}:{(player.current.length // 1000) % 60:02d}`** [{player.current.requester.mention}]"

        if player.queue.get_queue():
            tracks = [
                f"* [**`{track.title}`**]({track.uri}) by **{track.author}** - **`{track.length // 60000}:{(track.length // 1000) % 60:02d}`** [{track.requester.mention}]"
                for track in player.queue.get_queue()
            ]
            embeds = []
            queue_length = len(player.queue.get_queue())

            chunks = list(utils.as_chunks(tracks, 5))
            total_pages = len(chunks)
            for i, m in enumerate(chunks):
                embed = (
                    Embed(color=CONFIG.colors.base, description=playing)
                    .add_field(
                        name=f"Tracks ({queue_length})", value="\n".join([l for l in m])
                    )
                    .set_footer(text=f"{i+1}/{total_pages}")
                )
                embeds.append(embed)

            await ctx.paginator(embeds)
        else:
            embed = Embed(color=CONFIG.colors.base, description=playing)

            await ctx.send(embed=embed)

    @command(name="volume", aliases=["vol"])
    async def volume(self, ctx: Context, volume: int):
        """
        Set the volume of the player.
        """
        player: Player = ctx.voice_client
        max_volume = 200

        if volume < 0:
            return await ctx.warn("Volume cannot be negative.")

        if volume > max_volume:
            return await ctx.warn(f"Volume cannot exceed **`{max_volume}`**.")

        await player.set_volume(volume=volume)
        return await ctx.approve(f"Set the volume to **`{volume}`**")

    @command(name="seek")
    async def seek(self, ctx: Context, time: str):
        """
        Seek to a certain point in the song.
        """

        player: Player = ctx.voice_client

        try:
            minutes, seconds = map(int, time.split(":"))
            seek_time = minutes * 60 + seconds
        except ValueError:
            return await ctx.warn(
                "Invalid time format. Use `MM:SS` format. e.g.: `2:45`"
            )

        await player.seek(seek_time * 1000)
        return await ctx.approve(f"Seeked to **`{time}`**.")

    @command(name="speed")
    async def speed(self, ctx: Context, speed: float):
        """
        Change the speed of the song.
        """
        if speed < 0.50:
            return await ctx.warn("Speed cannot be less than **`0.50`**.")
        if speed > 5.00:
            return await ctx.warn("Speed cannot be more than **`5.00`**.")

        player: Player = ctx.voice_client
        try:
            await player.remove_filter(filter_tag="speed", fast_apply=True)
        except Exception:
            pass
        await player.add_filter(Timescale(speed=speed, tag="speed"), fast_apply=True)
        return await ctx.approve(f"Set the speed to **`{speed}`**.")

    @command(name="peak")
    async def peak(self, ctx: Context):
        """
        Show current track progress and info.
        """
        player: Player = ctx.voice_client

        if not player.is_playing:
            return await ctx.warn("Nothing is playing.")

        position = player.position
        duration = player.current_track.length
        percentage = position / duration

        current_time = f"{position // 60000}:{(position // 1000) % 60:02d}"
        total_time = f"{duration // 60000}:{(duration // 1000) % 60:02d}"

        progress_image = await self.create_progress_bar(percentage)

        embed = Embed(color=CONFIG.colors.base)
        embed.title = player.current_track.title
        embed.url = player.current_track.uri

        embed.description = f"By **{player.current_track.author}**\n"
        embed.description += f"**`{current_time}`** - **`{total_time}`**"

        if hasattr(player.current_track, "thumbnail"):
            embed.set_thumbnail(url=player.current_track.thumbnail)

        embed.set_footer(
            text=f"Requested by {player.current_track.requester.name}",
            icon_url=player.current_track.requester.display_avatar.url,
        )

        file = File(io.BytesIO(progress_image), filename="progress.png")
        embed.set_image(url="attachment://progress.png")

        await ctx.reply(embed=embed, file=file, allowed_mentions=AllowedMentions.none())

    async def create_progress_bar(self, percentage: float) -> bytes:
        """Create a progress bar image"""
        return await asyncio.to_thread(self._generate_progress_bar, percentage)

    def _generate_progress_bar(self, percentage: float) -> bytes:
        """Generate progress bar image synchronously"""
        width = 600
        height = 50
        padding = 12

        corner_radius = min(25, (height - 2 * padding) // 2, (width - 2 * padding) // 2)

        bot_color = CONFIG.colors.base.value
        r = (bot_color >> 16) & 255
        g = (bot_color >> 8) & 255
        b = bot_color & 255

        image = Image.new("RGBA", (width, height), (0, 0, 0, 0))
        draw = ImageDraw.Draw(image)

        def draw_rounded_rect(draw, x1, y1, x2, y2, radius, color):
            """Draw a rounded rectangle"""
            x1, x2 = sorted([x1, x2])
            y1, y2 = sorted([y1, y2])

            if radius > 0:
                draw.rectangle([x1 + radius, y1, x2 - radius, y2], fill=color)
                draw.rectangle([x1, y1 + radius, x2, y2 - radius], fill=color)
                draw.pieslice(
                    [x1, y1, x1 + 2 * radius, y1 + 2 * radius], 180, 270, fill=color
                )
                draw.pieslice(
                    [x2 - 2 * radius, y1, x2, y1 + 2 * radius], 270, 360, fill=color
                )
                draw.pieslice(
                    [x1, y2 - 2 * radius, x1 + 2 * radius, y2], 90, 180, fill=color
                )
                draw.pieslice(
                    [x2 - 2 * radius, y2 - 2 * radius, x2, y2], 0, 90, fill=color
                )
            else:
                draw.rectangle([x1, y1, x2, y2], fill=color)

        bg_color = (r // 4, g // 4, b // 4, 100)
        draw_rounded_rect(
            draw,
            padding,
            padding,
            width - padding,
            height - padding,
            corner_radius,
            bg_color,
        )

        progress_width = int((width - 2 * padding) * percentage)
        progress_width = max(0, min(progress_width, width - 2 * padding))

        progress_color = (r, g, b, 255)
        if progress_width > 0:
            draw_rounded_rect(
                draw,
                padding,
                padding,
                padding + progress_width,
                height - padding,
                corner_radius if progress_width >= corner_radius * 2 else 0,
                progress_color,
            )

        buffer = io.BytesIO()
        image.save(buffer, "PNG")
        buffer.seek(0)
        return buffer.getvalue()

    @command(name="autoplay", aliases=["ap"])
    async def autoplay(self, ctx: Context, *, amount: int = 5):
        """Enable autoplay."""
        player: Player = ctx.voice_client
        if not player or not player.is_playing:
            return await ctx.warn("There is no track playing right now.")

        if amount < 1 or amount > 15:
            return await ctx.warn("Please input an amount between 1 and 15.")

        current_track = player.current_track
        title, author = await self.clean_title_search(
            current_track.title, current_track.author
        )
        try:
            async with ctx.typing():
                url = (
                    "https://ws.audioscrobbler.com/2.0/?method=track.getsimilar"
                    f"&artist={author}"
                    f"&track={urllib.parse.quote_plus(title)}"
                    f"&api_key={CONFIG.lastfm.key}&format=json"
                )

                logger.info(url)

                async with self.bot.session.get(url) as response:
                    if not response.ok:
                        return await ctx.warn("Failed to fetch similar tracks.")

                    data = await response.json()
                    similar_tracks = data.get("similartracks", {}).get("track", [])

                added_count = 0
                for track in similar_tracks[:amount]:
                    track_title = track["name"]
                    track_artist = track["artist"]["name"]
                    search_query = f"{track_artist} - {track_title}"

                    results = await player.get_tracks(query=search_query, ctx=ctx)

                    if not results:
                        continue

                    track_to_add = results[0] if isinstance(results, list) else results

                    if not isinstance(track_to_add, Track):
                        continue

                    player.queue.put(track_to_add)
                    added_count += 1

                if added_count > 0:
                    await ctx.approve(f"Added **{added_count}** tracks to the queue.")
        except:
            url = (
                "https://ws.audioscrobbler.com/2.0/?method=chart.gettoptracks"
                f"&api_key={CONFIG.lastfm.key}&format=json"
            )
            async with self.bot.session.get(url) as response:
                if not response.ok:
                    return await ctx.warn("Failed to fetch popular tracks.")
                data = await response.json()
                similar_tracks = data.get("tracks", {}).get("track", [])

                for track in similar_tracks[:amount]:
                    track_title = track["name"]
                    track_artist = track["artist"]["name"]
                    search_query = f"{track_artist} - {track_title}"

                    results = await player.get_tracks(query=search_query, ctx=ctx)

                    if not results:
                        continue

                    track_to_add = results[0] if isinstance(results, list) else results

                    if not isinstance(track_to_add, Track):
                        continue

                    player.queue.put(track_to_add)
                    added_count += 1

                if added_count > 0:
                    await ctx.approve(f"Added **{added_count}** tracks to the queue.")
